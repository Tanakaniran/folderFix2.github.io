customer.controller.js
const db = require('../../config/database');

const getCustomers = async (request, reply) => {
    try {
        const { rows } = await db.query(
            "SELECT * FROM customers WHERE status != 'Archived' ORDER BY created_at DESC"
        );
        return reply.code(200).send({ success: true, data: rows });
    } catch (error) {
        request.log.error(error);
        return reply.code(500).send({ success: false, error: 'Gagal mengambil data pelanggan' });
    }
};

const createCustomer = async (request, reply) => {
    const { customer_id, name, email, phone, city, age, registration_date } = request.body;
    try {
        const { rows } = await db.query(
            `INSERT INTO customers (customer_id, name, email, phone, city, age, registration_date) 
             VALUES ($1, $2, $3, $4, $5, $6, $7) RETURNING *`,
            [customer_id, name, email, phone, city, age, registration_date]
        );
        return reply.code(201).send({ success: true, data: rows[0] });
    } catch (error) {
        return reply.code(400).send({ success: false, error: 'Gagal membuat pelanggan. Pastikan ID unik.' });
    }
};

const updateCustomerStatus = async (request, reply) => {
    const { id } = request.params;
    const { status } = request.body;
    try {
        const { rows } = await db.query(
            "UPDATE customers SET status = $1, updated_at = CURRENT_TIMESTAMP WHERE id = $2 RETURNING *",
            [status, id]
        );
        if (rows.length === 0) return reply.code(404).send({ error: 'Pelanggan tidak ditemukan' });
        return reply.code(200).send({ success: true, data: rows[0] });
    } catch (error) {
        return reply.code(500).send({ success: false, error: 'Gagal memperbarui status' });
    }
};

module.exports = { getCustomers, createCustomer, updateCustomerStatus };

customer.routes.js
const customerController = require('./customers.controller');

async function customerRoutes(fastify, options) {
    fastify.get('/', customerController.getCustomers);
    fastify.post('/', customerController.createCustomer);
    fastify.patch('/:id/status', customerController.updateCustomerStatus);
}

module.exports = customerRoutes;
